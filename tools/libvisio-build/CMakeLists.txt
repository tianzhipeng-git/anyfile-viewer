cmake_minimum_required(VERSION 3.16)
project(anyfile_visio C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -fexceptions -sUSE_BOOST_HEADERS=1 -sUSE_ICU=1 -sUSE_ZLIB=1")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O2")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_PROGRAMS OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_PYTHON OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_ICONV OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_LZMA OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_ZLIB OFF CACHE BOOL "" FORCE)
add_subdirectory(${SOURCE_ROOT}/libxml2-2.15.4 xml)
set(REVENGE ${SOURCE_ROOT}/librevenge-0.0.5)
set(VISIO ${SOURCE_ROOT}/libvisio-0.1.11)
set(REVENGE_FILES RVNGBinaryData RVNGMemoryStream RVNGProperty RVNGPropertyList RVNGPropertyListVector RVNGString RVNGStringVector RVNGSVGDrawingGenerator RVNGStreamImplementation RVNGOLEStream RVNGZipStream RVNGDirectoryStream)
foreach(name IN LISTS REVENGE_FILES)
  list(APPEND REVENGE_SOURCES ${REVENGE}/src/lib/${name}.cpp)
endforeach()
file(GLOB VISIO_SOURCES ${VISIO}/src/lib/*.cpp)
add_executable(visio ${CMAKE_CURRENT_SOURCE_DIR}/adapter.cpp ${REVENGE_SOURCES} ${VISIO_SOURCES})
target_include_directories(visio PRIVATE ${REVENGE}/inc ${VISIO}/inc ${VISIO}/src/lib)
target_compile_definitions(visio PRIVATE LIBREVENGE_BUILD LIBREVENGE_STREAM_BUILD LIBVISIO_BUILD BOOST_ERROR_CODE_HEADER_ONLY BOOST_SPIRIT_USE_PHOENIX_V3 BOOST_SYSTEM_NO_DEPRECATED NDEBUG)
target_link_libraries(visio PRIVATE LibXml2)
set_target_properties(visio PROPERTIES SUFFIX ".mjs")
target_link_options(visio PRIVATE --bind -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912 -sFILESYSTEM=0 -sUSE_ICU=1 -sUSE_ZLIB=1)
